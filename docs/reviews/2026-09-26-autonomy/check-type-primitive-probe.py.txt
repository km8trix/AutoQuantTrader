def _original(value: object, annotation: object, name: str) -> None:
    if isinstance(annotation, TypeAliasType):
        _original(value, annotation.__value__, name)
        return
    origin, args = get_origin(annotation), get_args(annotation)
    if origin in (types.UnionType, Union):
        for candidate in args:
            try:
                _original(value, candidate, name)
                return
            except (TypeError, ValueError):
                pass
        raise ValueError(f"{name} has an unsupported value type")
    if origin is Literal:
        if not any(type(value) is type(v) and value == v for v in args):
            raise ValueError(f"{name} has an unsupported literal")
        return
    if origin is tuple:
        if type(value) is not tuple:
            raise ValueError(f"{name} must be immutable tuple")
        if len(args) == 2 and args[1] is Ellipsis:
            for item in value:
                _original(item, args[0], name)
        elif len(value) == len(args):
            for item, expected in zip(value, args, strict=True):
                _original(item, expected, name)
        else:
            raise ValueError(f"{name} tuple shape differs")
        return
    if annotation is type(None):
        if value is not None:
            raise ValueError(f"{name} must be null")
        return
    if not isinstance(annotation, type) or type(value) is not annotation:
        raise ValueError(f"{name} has an unsupported value type")
    if type(value) is Decimal and not value.is_finite():
        raise ValueError(f"{name} must be finite")
    if type(value) is datetime:
        require_utc(value, name)
    if type(value) is str and len(value) > 65536:
        raise ValueError(f"{name} exceeds text bound")

class ObservedType(type):
    def __getattribute__(cls, name):
        if name == "__class__":
            trace.append("annotation.__class__")
        return super().__getattribute__(name)

    def __eq__(cls, other):
        trace.append("annotation.__eq__")
        return super().__eq__(other)

    __hash__ = type.__hash__


class ObservedValue(metaclass=ObservedType):
    pass


sentinel = StopIteration("literal annotation failure")


class FailingAnnotation:
    @property
    def __class__(self):
        trace.append("annotation.__class__.raise")

class ObservedInt(int):

def outcome(function, value, annotation):
    trace.clear()
    try:
        result = function(value, annotation, "sample")
    except BaseException as error:
        result = {
            "status": "error", "type": type(error).__name__, "message": str(error),
            "cause": None if error.__cause__ is None else type(error.__cause__).__name__,
            "context": None if error.__context__ is None else type(error.__context__).__name__,
            "is_sentinel": error is sentinel,
            "cause_is_sentinel": error.__cause__ is sentinel,
        }
        # The same fixed sentinel is intentionally reused; do not retain its traceback.
        sentinel.__traceback__ = None
    else:
        result = {"status": "returned", "is_none": result is None}
    return {**result, "trace": trace.copy()}


compatibility = []
for label, value, annotation in cases:
    expected = outcome(_original, value, annotation)
    actual = outcome(_candidate, value, annotation)
    # Compare project original independently as well, including standard typing helpers.
    project = outcome(contracts._check_type, value, annotation)
    compatibility.append({"case": label, "matches": expected == actual == project,
                          "original": expected, "candidate": actual, "project": project})

primitive = [("short", str), (7, int), (False, bool), (None, type(None))] * 12
fallback = [(Decimal("1.25"), Decimal), (datetime(2020, 1, 1, tzinfo=UTC), datetime),
            (Count(7), Count), (Status.OK, Status), ((1, 2), tuple[int, ...]),
            ("ok", Literal["ok", "no"])] * 8
mixed = primitive[:24] + fallback[:24]
# Literal fields of the existing foundational VersionPin, with no reconstruction or I/O.
pin_fields = [("name", str), ("v1", str), ("a" * 64, str)] * 16
workloads = {"matching_primitives": primitive, "fallback_control": fallback,
             "mixed_synthetic": mixed, "version_pin_declared_string_fields": pin_fields}
iterations, repeats = 300, 7
measurements = []
for label, workload in workloads.items():
    values = {"original": [], "candidate": []}
    for repeat in range(repeats):
        order = (("original", _original), ("candidate", _candidate))
        if repeat % 2:
            order = tuple(reversed(order))
        for name, function in order:
            start = time.perf_counter_ns()
            for _ in range(iterations):
                for value, annotation in workload:
                    function(value, annotation, "sample")
            values[name].append(time.perf_counter_ns() - start)
    original_median = statistics.median(values["original"])
    candidate_median = statistics.median(values["candidate"])
    measurements.append({"workload": label, "top_level_calls_per_sample": iterations * len(workload),
                         "times_ns": values, "median_ns": {"original": original_median, "candidate": candidate_median},
                         "candidate_over_original": candidate_median / original_median})
