class Current:
    def __init__(self, children=()):
        self.children = children

        self.builder_error = self.public_error = None


HALTED = object()


class Store:
    def __init__(self, current):
        self.before = tuple(current.children)

    @staticmethod
    def _assignment_identity_fields(value):
        seen = {id(value), id(value.children)}
        if value.builder_error is not None:
            raise value.builder_error
        return tuple(value.children)

    def require_resolved_snapshot(self, value):
        before = self.before
        current = self._assignment_identity_fields(value)
        if value.public_error is not None:
            raise value.public_error
        assert len(before) == len(current)
        assert all(a is b for a, b in zip(before, current, strict=True))

class Operation(SimpleNamespace):

class Reader:
    def __init__(self, current, proof=True):
        self.current = current
        self.daily = Store(current)
        self.composer = object()
        self._factory_read = Operation(reader=self, thread=threading.current_thread(),
                                      failed=False, scope=object())
        episode = Episode(reader=self, composer=self.composer, current=current,
                          thread=threading.current_thread())
        self._daily_episode = State(episode=episode, failed=False, fingerprint_context=None)
        self.proof = proof

    def _require_daily_graph(self, current, use=None):
        self.daily.require_resolved_snapshot(current)
        self.daily.require_resolved_snapshot(current)

    def _require_borrow_daily_graph(self, current, borrow):
        self._require_daily_graph(current, borrow)

    def _require_daily_episode(self, episode, borrow=None):
        if borrow is None:
            self._require_daily_graph(episode.current)
        else:
            self._require_borrow_daily_graph(episode.current, borrow)

    def _borrow_original_daily(self, episode, *, consumer):
        for _ in range(3):
            self._require_daily_episode(episode, object() if self.proof else None)
        return episode.current


class Factory:
    def __init__(self, reader):
        self.integrity = reader
        self.extra = None

    def execute(self, *, operation_id):
        if self.extra is not None:
            self.extra()
        reader = self.integrity
        return reader._borrow_original_daily(reader._daily_episode.episode, consumer=reader.composer)

def run(capture, factory):
    capture.start()
    try:
        return factory.execute(operation_id=plugin.OPERATION)
    finally:
        capture.stop()


def clean(capture):

    assert sys.monitoring.get_tool(plugin.TOOL_ID) is None
    assert sys.monitoring.get_events(plugin.TOOL_ID) == 0
    for code in capture.codes:

    assert capture.factory_ref is capture.reader_ref is capture.store_ref is capture.current_ref is None
