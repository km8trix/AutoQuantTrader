def load(mode):
    node = copy.deepcopy(original_ast)
    node.decorator_list = []
    loop = next(item for item in node.body if isinstance(item, ast.While))
    index = next(index for index, item in enumerate(loop.body) if isinstance(item, ast.If)
                 and isinstance(item.test, ast.BoolOp)
                 and 'item_type is str' in ast.unparse(item.test))
    old = loop.body[index]
    if mode != 'original':
        members = '_SCALARS' if mode == 'frozen_native_set' else '(str, int, bytes, type(None), bool, Decimal, datetime, date, object)'
        inner = ast.If(test=ast.parse(f'item_type in {members}', mode='eval').body,
                       body=copy.deepcopy(old.body), orelse=[])
        loop.body[index] = ast.If(test=ast.parse('type(item_type) is type', mode='eval').body,
                                  body=[inner], orelse=[old])
    module = ast.fix_missing_locations(ast.Module(body=[node], type_ignores=[]))
    namespace = dict(Mapping=Mapping, fields=fields, is_dataclass=is_dataclass,
                     Decimal=Decimal, date=date, datetime=datetime, _SCALARS=frozenset(SCALARS))
    exec(compile(module, '<finite-daily-identity-probe>', 'exec'), namespace)
    return namespace['_assignment_identity_fields'], namespace


def outcome(function, value, trace):
    try:
        result = function(value)
    except BaseException as error:
        return {'status': 'error', 'type': type(error).__name__,
                'cause_type': None if error.__cause__ is None else type(error.__cause__).__name__,
                'context_type': None if error.__context__ is None else type(error.__context__).__name__,
                'trace': list(trace)}
    return {'status': 'returned', 'identity_count': len(result), 'trace': list(trace)}


def disappearing_alias(mode):
    function, namespace = load(mode)
    trace = []
    @dataclass
    class Record:
        field: object
        def __getattribute__(self, name):
            if name == 'field':
                trace.append('field')
                del namespace['Decimal']
            return object.__getattribute__(self, name)
    return outcome(function, Record('exact string leaf'), trace)


def rebound_alias(mode):
    function, namespace = load(mode)
    trace = []
    @dataclass
    class NewScalar:
        field: object
        def __getattribute__(self, name):
            if name == 'field':
                trace.append('unexpected child getter')
                raise ValueError('toy callback must be skipped by original live alias')
            return object.__getattribute__(self, name)
    @dataclass
    class Root:
        field: object
        def __getattribute__(self, name):
            if name == 'field':
                trace.append('root getter')
                namespace['Decimal'] = NewScalar
            return object.__getattribute__(self, name)
    return outcome(function, Root(NewScalar('child')), trace)


def custom_meta(mode):
    function, namespace = load(mode)
    trace = []
    class Meta(type):
        def __hash__(cls):
            trace.append('custom hash')
            raise AssertionError
        def __eq__(cls, other):
            trace.append('custom equality')
            raise AssertionError
        def __getattribute__(cls, name):
            if name == '__dataclass_fields__':
                trace.append('original reflection')
            return super().__getattribute__(name)
    class Unknown(metaclass=Meta):
        pass
    return outcome(function, (Unknown(),), trace)
