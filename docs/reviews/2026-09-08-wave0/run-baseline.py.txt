def run(name, args, cwd=R, timeout=300):
    start=time.monotonic()
    with (E/'logs'/f'{name}.log').open('w') as log:
        try:
            result=subprocess.run(args, cwd=cwd, env=ENV, stdout=log, stderr=subprocess.STDOUT, timeout=timeout)
            status=result.returncode
        except subprocess.TimeoutExpired:
            status='timeout'
    record={'name':name,'command':args,'cwd':str(cwd),'exit_code':status,'seconds':round(time.monotonic()-start,3),'log':f'logs/{name}.log'}
    (E/f'{name}.json').write_text(json.dumps(record,indent=2)+'\n')
    lines=(E/'logs'/f'{name}.log').read_text(errors='replace').splitlines()
    print(json.dumps(record),flush=True)
    print('\n'.join(lines[-12:]),flush=True)
